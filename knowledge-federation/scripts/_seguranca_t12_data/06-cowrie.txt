  jq --unbuffered 'select(.eventid == "cowrie.session.connect" or .eventid == "cowrie.login.success" or .eventid == "cowrie.login.failed") | {time: .timestamp, event: .eventid, attacker_ip: .src_ip, user: .username, pass: .password}'
