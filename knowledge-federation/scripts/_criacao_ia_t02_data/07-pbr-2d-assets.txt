from PIL import Image

def build_spritesheet(frames: list[Image.Image], frame_size: int = 64) -> Image.Image:
    sheet = Image.new("RGBA", (frame_size * len(frames), frame_size), (0, 0, 0, 0))
    for i, frame in enumerate(frames):
